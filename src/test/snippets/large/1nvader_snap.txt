; patches the game: with the trainer, invader_row's increment calls trainer_next_row;
; without it, title_screen calls save_highscore
; called by: crack_intro
apply_trainer:
		ror trainer_flag
		bcc lC0DB
		lda #$ea
		sta row_add_right
		sta row_add_right + 1
		sta row_add_left
		sta row_add_left + 1
		lda #$20
		sta row_store_right
		sta row_store_left
		lda #$eb
		sta row_store_right + 1
		sta row_store_left + 1
		lda #$c0
		sta row_store_right + 2
		sta row_store_left + 2
		rts

		lda #$20
		sta title_screen
		lda #$48
		sta title_screen + 1
		lda #$c1
		sta title_screen + 2
		rts
