; ---- crack intro and high-score code ----

		cli 
		lda #$fc
		sta stop_vector
		lsr 
		sta irq_vector
		lda #$80
		sta kernal_case_lock
		asl 
		sta border_color
		sta background_color
		tax 
; jumped to by: crack_intro
lC017:
		sta color_ram, x
		sta color_ram + $100, x
		sta color_ram + $200, x
		sta color_ram + $300, x
		inx 
		bne lC017
; jumped to by: crack_intro
lC026:
		lda title_line, x
		sta screen, x
		lda question_line, x
		sta screen + $c8, x
		lda #$0f
		sta color_ram, x
		sta color_ram + $c8, x
		inx 
		cpx #$28
		bne lC026
		jsr ask_trainer
		sty cia1_port_a
		jsr load_highscore
		jsr apply_trainer
		jmp start

; patches the game: with the trainer, invader_row's increment calls trainer_next_row;
; without it, title_screen calls save_highscore
; called by: crack_intro
apply_trainer:
		ror trainer_flag
		bcc lC0DB
		lda #$ea
		sta row_add_right
		sta row_add_right + 1
		sta row_add_left
		sta row_add_left + 1
		lda #$20
		sta row_store_right
		sta row_store_left
		lda #$eb
		sta row_store_right + 1
		sta row_store_left + 1
		lda #$c0
		sta row_store_right + 2
		sta row_store_left + 2
		rts 

; jumped to by: apply_trainer
lC0DB:
		lda #$20
		sta title_screen
		lda #$48
		sta title_screen + 1
		lda #$c1
		sta title_screen + 2
		rts
